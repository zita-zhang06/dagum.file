from pathlib import Path

import numpy as np
import pandas as pd
from openpyxl.styles import Alignment, Font, PatternFill
from scipy.stats import burr


def find_data_file():
    names = (
        "EMDAT(2)(1).xlsx.xlsx",
        "EMDAT(2)(1).xlsx",
        "EMDAT.xlsx",
    )

    for folder in (Path.cwd(), Path.home() / "Desktop"):
        for name in names:
            path = folder / name
            if path.exists():
                return path

    raise FileNotFoundError(
        "Put EMDAT(2)(1).xlsx.xlsx on the Desktop "
        "or in the current folder."
    )

DATA_FILE = find_data_file()

OUTPUT_FILE = (
    DATA_FILE.parent
    / "Table_2_Baseline_Monte_Carlo_Results.xlsx"
)

DAMAGE_COLUMN = "Total Damage ('000 US$)"
GDP_COLUMN = "GDP (current US$)"

FLOOD_SUBTYPES = (
    "Flash flood",
    "Flood (General)",
    "Riverine flood",
)

PARETO_THRESHOLD = 2e-5

events = pd.read_excel(
    DATA_FILE,
    sheet_name="Data all",
)

gdp = pd.read_excel(
    DATA_FILE,
    sheet_name="GDP",
)

events.columns = events.columns.astype(str).str.strip()
gdp.columns = gdp.columns.astype(str).str.strip()

events["Start Year"] = pd.to_numeric(
    events["Start Year"],
    errors="coerce",
)

events[DAMAGE_COLUMN] = pd.to_numeric(
    events[DAMAGE_COLUMN],
    errors="coerce",
)

gdp["Year"] = pd.to_numeric(
    gdp["Year"],
    errors="coerce",
)

gdp[GDP_COLUMN] = pd.to_numeric(
    gdp[GDP_COLUMN],
    errors="coerce",
)

mask = (
    events["Country"].astype(str).str.strip().eq("Brazil")
    & events["Disaster Type"].astype(str).str.strip().eq("Flood")
    & events["Disaster Subtype"]
        .astype(str)
        .str.strip()
        .isin(FLOOD_SUBTYPES)
    & events["Start Year"].between(2000, 2024)
    & events[DAMAGE_COLUMN].notna()
    & events[DAMAGE_COLUMN].gt(0)
)

events = events.loc[mask].copy()
events["Year"] = events["Start Year"].astype(int)

gdp = gdp.loc[
    gdp["Year"].between(2000, 2024),
    ["Year", GDP_COLUMN],
].copy()

gdp["Year"] = gdp["Year"].astype(int)

data = events.merge(
    gdp,
    on="Year",
    how="left",
    validate="many_to_one",
)

if data[GDP_COLUMN].isna().any():
    raise ValueError(
        "GDP is missing for one or more event years."
    )

data["damage_ratio"] = (
    data[DAMAGE_COLUMN] * 1000.0 / data[GDP_COLUMN]
)

x_dagum = np.sort(
    data.loc[
        data["damage_ratio"] > 0,
        "damage_ratio",
    ].to_numpy(float)
)

x_pareto = np.sort(
    x_dagum[x_dagum >= PARETO_THRESHOLD]
)

n_dagum = len(x_dagum)
n_pareto = len(x_pareto)

if (n_dagum, n_pareto) != (51, 38):
    raise ValueError(
        f"Expected 51 Dagum and 38 Pareto observations; "
        f"got {n_dagum} and {n_pareto}."
    )

lambda_dagum = n_dagum / 25
lambda_pareto = n_pareto / 25

pareto_alpha = n_pareto / np.sum(
    np.log(x_pareto / PARETO_THRESHOLD)
)

dagum_a, dagum_p, _, dagum_b = burr.fit(
    x_dagum,
    floc=0,
)

years = 50
dt = 1 / 12
n_steps = int(years / dt)

n_simulations = 10000

# Baseline rebuilding rate: theta = 1.0 per year
recovery_factor = 1 - 1.0 * dt

# Collapse threshold: 1% of capital
collapse_threshold = 0.01

# Persistent collapse is evaluated over the final five years.
last_window_steps = 60
last_window_start = n_steps - last_window_steps

def simulate(model, arrival_rate, seed):
    rng = np.random.default_rng(seed)

    counts = rng.poisson(
        arrival_rate * dt,
        size=(n_simulations, n_steps),
    ).astype(np.int16)

    flat = counts.ravel()

    indices = np.repeat(
        np.arange(flat.size, dtype=np.int32),
        flat,
    )

    u = np.clip(
        rng.random(int(flat.sum())),
        1e-12,
        1 - 1e-12,
    )

    if model == "Pareto":
        losses = (
            PARETO_THRESHOLD
            * (1 - u) ** (-1 / pareto_alpha)
        )

    else:
        losses = (
            dagum_b
            * (u ** (-1 / dagum_p) - 1)
            ** (-1 / dagum_a)
        )

    jumps = np.bincount(
        indices,
        weights=losses,
        minlength=flat.size,
    ).reshape(n_simulations, n_steps)

    current = np.zeros(n_simulations)
    maximum = np.zeros(n_simulations)

    transient = np.zeros(
        n_simulations,
        dtype=bool,
    )

    final_window_sum = np.zeros(
        n_simulations
    )

    for step in range(n_steps):

        current = np.clip(
            current * recovery_factor
            + jumps[:, step],
            0,
            1,
        )

        maximum = np.maximum(
            maximum,
            current,
        )

        transient |= (
            current >= collapse_threshold
        )

        if step >= last_window_start:
            final_window_sum += current

    final_window_average = (
        final_window_sum / last_window_steps
    )

    return (
        current,
        maximum,
        transient,
        final_window_average,
    )

final_p, maximum_p, transient_p, last_average_p = simulate(
    model="Pareto",
    arrival_rate=lambda_pareto,
    seed=12345,
)

final_d, maximum_d, transient_d, last_average_d = simulate(
    model="Dagum",
    arrival_rate=lambda_dagum,
    seed=54321,
)

def summary(
    final,
    maximum,
    transient,
    last_average,
):
    return [
        np.mean(final) * 100,
        np.median(final) * 100,
        np.mean(maximum) * 100,
        np.median(maximum) * 100,
        np.percentile(maximum, 95) * 100,
        np.percentile(maximum, 99) * 100,
        np.mean(transient) * 100,
        np.mean(
            final >= collapse_threshold
        ) * 100,
        np.mean(
            last_average >= collapse_threshold
        ) * 100,
        np.mean(
            maximum >= 1.0
        ) * 100,
    ]


measures = [
    "Mean final damage (%)",
    "Median final damage (%)",
    "Mean maximum damage (%)",
    "Median maximum damage (%)",
    "95th percentile maximum damage (%)",
    "99th percentile maximum damage (%)",
    "Transient threshold exceedance probability (%)",
    "Final-year threshold exceedance probability (%)",
    "Persistent collapse probability (%)",
    "Probability of reaching the 100% cap (%)",
]

table = pd.DataFrame(
    {
        "Measure": measures,

        "Pareto": summary(
            final_p,
            maximum_p,
            transient_p,
            last_average_p,
        ),

        "Dagum": summary(
            final_d,
            maximum_d,
            transient_d,
            last_average_d,
        ),
    }
)

with pd.ExcelWriter(
    OUTPUT_FILE,
    engine="openpyxl",
) as writer:

    table.to_excel(
        writer,
        sheet_name="Table 2",
        index=False,
    )

    ws = writer.book["Table 2"]

    for cell in ws[1]:
        cell.font = Font(bold=True)

        cell.fill = PatternFill(
            "solid",
            fgColor="D9D9D9",
        )

        cell.alignment = Alignment(
            horizontal="center"
        )

    ws.column_dimensions["A"].width = 54
    ws.column_dimensions["B"].width = 18
    ws.column_dimensions["C"].width = 18

    for row in ws.iter_rows(
        min_row=2,
        min_col=2,
        max_col=3,
    ):
        for cell in row:
            cell.number_format = "0.00"

            cell.alignment = Alignment(
                horizontal="center"
            )

print(
    table.round(2).to_string(index=False)
)

print()
print("Saved:", OUTPUT_FILE)
