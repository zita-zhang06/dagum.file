from pathlib import Path

import numpy as np
import pandas as pd
from openpyxl.styles import Alignment, Font, PatternFill
from scipy.stats import burr


def find_data_file():
    names = (
        "EMDAT(2)(1).xlsx.xlsx",
        "EMDAT(2)(1).xlsx",
        "EMDAT.xlsx",
    )

    for folder in (Path.cwd(), Path.home() / "Desktop"):
        for name in names:
            path = folder / name

            if path.exists():
                return path

    raise FileNotFoundError(
        "Put EMDAT(2)(1).xlsx.xlsx on the Desktop "
        "or in the current folder."
    )

DATA_FILE = find_data_file()

OUTPUT_FILE = (
    DATA_FILE.parent
    / "Table_3_Selected_Persistent_Collapse_Results.xlsx"
)

DAMAGE_COLUMN = "Total Damage ('000 US$)"
GDP_COLUMN = "GDP (current US$)"

FLOOD_SUBTYPES = (
    "Flash flood",
    "Flood (General)",
    "Riverine flood",
)

PARETO_THRESHOLD = 2e-5

events = pd.read_excel(
    DATA_FILE,
    sheet_name="Data all",
)

gdp = pd.read_excel(
    DATA_FILE,
    sheet_name="GDP",
)

events.columns = events.columns.astype(str).str.strip()
gdp.columns = gdp.columns.astype(str).str.strip()

events["Start Year"] = pd.to_numeric(
    events["Start Year"],
    errors="coerce",
)

events[DAMAGE_COLUMN] = pd.to_numeric(
    events[DAMAGE_COLUMN],
    errors="coerce",
)

gdp["Year"] = pd.to_numeric(
    gdp["Year"],
    errors="coerce",
)

gdp[GDP_COLUMN] = pd.to_numeric(
    gdp[GDP_COLUMN],
    errors="coerce",
)

mask = (
    events["Country"].astype(str).str.strip().eq("Brazil")
    & events["Disaster Type"].astype(str).str.strip().eq("Flood")
    & events["Disaster Subtype"]
        .astype(str)
        .str.strip()
        .isin(FLOOD_SUBTYPES)
    & events["Start Year"].between(2000, 2024)
    & events[DAMAGE_COLUMN].notna()
    & events[DAMAGE_COLUMN].gt(0)
)

events = events.loc[mask].copy()
events["Year"] = events["Start Year"].astype(int)

gdp = gdp.loc[
    gdp["Year"].between(2000, 2024),
    ["Year", GDP_COLUMN],
].copy()

gdp["Year"] = gdp["Year"].astype(int)

data = events.merge(
    gdp,
    on="Year",
    how="left",
    validate="many_to_one",
)

if data[GDP_COLUMN].isna().any():
    raise ValueError(
        "GDP is missing for one or more event years."
    )

data["damage_ratio"] = (
    data[DAMAGE_COLUMN] * 1000.0 / data[GDP_COLUMN]
)

x_dagum = np.sort(
    data.loc[
        data["damage_ratio"] > 0,
        "damage_ratio",
    ].to_numpy(float)
)

x_pareto = np.sort(
    x_dagum[x_dagum >= PARETO_THRESHOLD]
)

n_dagum = len(x_dagum)
n_pareto = len(x_pareto)

if (n_dagum, n_pareto) != (51, 38):
    raise ValueError(
        f"Expected 51 Dagum and 38 Pareto observations; "
        f"got {n_dagum} and {n_pareto}."
    )

lambda_dagum = n_dagum / 25
lambda_pareto = n_pareto / 25

pareto_alpha = n_pareto / np.sum(
    np.log(x_pareto / PARETO_THRESHOLD)
)

dagum_a, dagum_p, _, dagum_b = burr.fit(
    x_dagum,
    floc=0,
)

years = 50
dt = 1 / 12
n_steps = int(years / dt)

n_simulations = 5000

theta_grid = np.round(
    np.arange(0.1, 3.01, 0.1),
    2,
)

collapse_threshold = 0.01

# Final five years = 60 monthly observations
last_window_steps = 60
last_window_start = n_steps - last_window_steps

def generate_jumps(model, arrival_rate, seed):
    rng = np.random.default_rng(seed)

    counts = rng.poisson(
        arrival_rate * dt,
        size=(n_simulations, n_steps),
    ).astype(np.int16)

    flat = counts.ravel()

    indices = np.repeat(
        np.arange(flat.size, dtype=np.int32),
        flat,
    )

    u = np.clip(
        rng.random(int(flat.sum())),
        1e-12,
        1 - 1e-12,
    )

    if model == "Pareto":
        losses = (
            PARETO_THRESHOLD
            * (1 - u) ** (-1 / pareto_alpha)
        )

    else:
        losses = (
            dagum_b
            * (u ** (-1 / dagum_p) - 1)
            ** (-1 / dagum_a)
        )

    return np.bincount(
        indices,
        weights=losses,
        minlength=flat.size,
    ).reshape(n_simulations, n_steps)


jumps_p = generate_jumps(
    model="Pareto",
    arrival_rate=lambda_pareto,
    seed=24680,
)

jumps_d = generate_jumps(
    model="Dagum",
    arrival_rate=lambda_dagum,
    seed=13579,
)

# Identical model-specific shocks are reused across theta values.
recovery_factors = (
    1 - theta_grid * dt
)[:, None]

current_p = np.zeros(
    (len(theta_grid), n_simulations)
)

current_d = np.zeros(
    (len(theta_grid), n_simulations)
)

last_sum_p = np.zeros_like(current_p)
last_sum_d = np.zeros_like(current_d)

for step in range(n_steps):

    current_p = np.clip(
        current_p * recovery_factors
        + jumps_p[:, step][None, :],
        0,
        1,
    )

    current_d = np.clip(
        current_d * recovery_factors
        + jumps_d[:, step][None, :],
        0,
        1,
    )

    if step >= last_window_start:
        last_sum_p += current_p
        last_sum_d += current_d

pareto_probability = (
    np.mean(
        last_sum_p / last_window_steps
        >= collapse_threshold,
        axis=1,
    )
    * 100
)

dagum_probability = (
    np.mean(
        last_sum_d / last_window_steps
        >= collapse_threshold,
        axis=1,
    )
    * 100
)

full = pd.DataFrame(
    {
        "Rebuilding rate, θ": theta_grid,
        "Pareto probability (%)": pareto_probability,
        "Dagum probability (%)": dagum_probability,
    }
)

selected_rates = [
    0.1,
    0.5,
    1.0,
    1.5,
    2.0,
    2.5,
    3.0,
]

selected = full.loc[
    full["Rebuilding rate, θ"].isin(selected_rates)
].copy()

selected["Difference (percentage points)"] = (
    selected["Pareto probability (%)"]
    - selected["Dagum probability (%)"]
)

pareto_crossing = np.where(
    pareto_probability <= 5.0
)[0]

dagum_crossing = np.where(
    dagum_probability <= 5.0
)[0]

critical = pd.DataFrame(
    {
        "Model": [
            "Pareto",
            "Dagum",
        ],

        "5% probability benchmark": [
            "≤ 5%",
            "≤ 5%",
        ],

        "Critical rebuilding rate, θcrit": [
            (
                theta_grid[pareto_crossing[0]]
                if pareto_crossing.size
                else np.nan
            ),
            (
                theta_grid[dagum_crossing[0]]
                if dagum_crossing.size
                else np.nan
            ),
        ],
    }
)


with pd.ExcelWriter(
    OUTPUT_FILE,
    engine="openpyxl",
) as writer:

    selected.to_excel(
        writer,
        sheet_name="Table 3",
        index=False,
        startrow=0,
    )

    critical.to_excel(
        writer,
        sheet_name="Table 3",
        index=False,
        startrow=len(selected) + 3,
    )

    ws = writer.book["Table 3"]

    header_rows = [
        1,
        len(selected) + 4,
    ]

    for row_number in header_rows:
        for cell in ws[row_number]:

            if cell.value is not None:
                cell.font = Font(bold=True)

                cell.fill = PatternFill(
                    "solid",
                    fgColor="D9D9D9",
                )

                cell.alignment = Alignment(
                    horizontal="center",
                    wrap_text=True,
                )

    widths = {
        "A": 24,
        "B": 26,
        "C": 26,
        "D": 32,
    }

    for column, width in widths.items():
        ws.column_dimensions[column].width = width

    for row in ws.iter_rows(min_row=2):
        for cell in row:

            cell.alignment = Alignment(
                horizontal="center",
                wrap_text=True,
            )

            if isinstance(cell.value, (float, int)):
                cell.number_format = "0.00"


print(
    selected.round(2).to_string(index=False)
)

print()

print(
    critical.to_string(index=False)
)

print()
print("Saved:", OUTPUT_FILE)
