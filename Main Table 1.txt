from pathlib import Path
import warnings

import numpy as np
import pandas as pd
from openpyxl.styles import Alignment, Font, PatternFill
from scipy.stats import burr


def find_data_file():
    names = (
        "EMDAT(2)(1).xlsx.xlsx",
        "EMDAT(2)(1).xlsx",
        "EMDAT.xlsx",
    )

    for folder in (Path.cwd(), Path.home() / "Desktop"):
        for name in names:
            path = folder / name
            if path.exists():
                return path

    raise FileNotFoundError(
        "Put EMDAT(2)(1).xlsx.xlsx on the Desktop "
        "or in the current folder."
    )

DATA_FILE = find_data_file()

OUTPUT_FILE = (
    DATA_FILE.parent
    / "Table_1_Frequency_Severity_Estimates.xlsx"
)

DAMAGE_COLUMN = "Total Damage ('000 US$)"
GDP_COLUMN = "GDP (current US$)"

FLOOD_SUBTYPES = (
    "Flash flood",
    "Flood (General)",
    "Riverine flood",
)

PARETO_THRESHOLD = 2e-5

BOOTSTRAP_REPLICATIONS = 1000
BOOTSTRAP_SEED = 20260713

events = pd.read_excel(
    DATA_FILE,
    sheet_name="Data all",
)

gdp = pd.read_excel(
    DATA_FILE,
    sheet_name="GDP",
)

events.columns = events.columns.astype(str).str.strip()
gdp.columns = gdp.columns.astype(str).str.strip()

events["Start Year"] = pd.to_numeric(
    events["Start Year"],
    errors="coerce",
)

events[DAMAGE_COLUMN] = pd.to_numeric(
    events[DAMAGE_COLUMN],
    errors="coerce",
)

gdp["Year"] = pd.to_numeric(
    gdp["Year"],
    errors="coerce",
)

gdp[GDP_COLUMN] = pd.to_numeric(
    gdp[GDP_COLUMN],
    errors="coerce",
)

mask = (
    events["Country"].astype(str).str.strip().eq("Brazil")
    & events["Disaster Type"].astype(str).str.strip().eq("Flood")
    & events["Disaster Subtype"]
        .astype(str)
        .str.strip()
        .isin(FLOOD_SUBTYPES)
    & events["Start Year"].between(2000, 2024)
    & events[DAMAGE_COLUMN].notna()
    & events[DAMAGE_COLUMN].gt(0)
)

events = events.loc[mask].copy()
events["Year"] = events["Start Year"].astype(int)

gdp = gdp.loc[
    gdp["Year"].between(2000, 2024),
    ["Year", GDP_COLUMN],
].copy()

gdp["Year"] = gdp["Year"].astype(int)

data = events.merge(
    gdp,
    on="Year",
    how="left",
    validate="many_to_one",
)

if data[GDP_COLUMN].isna().any():
    raise ValueError(
        "GDP is missing for one or more event years."
    )

data["damage_ratio"] = (
    data[DAMAGE_COLUMN] * 1000.0 / data[GDP_COLUMN]
)

x_dagum = np.sort(
    data.loc[
        data["damage_ratio"] > 0,
        "damage_ratio",
    ].to_numpy(float)
)

x_pareto = np.sort(
    x_dagum[x_dagum >= PARETO_THRESHOLD]
)

n_dagum = len(x_dagum)
n_pareto = len(x_pareto)

if (n_dagum, n_pareto) != (51, 38):
    raise ValueError(
        f"Expected 51 Dagum and 38 Pareto observations; "
        f"got {n_dagum} and {n_pareto}."
    )

pareto_alpha = n_pareto / np.sum(
    np.log(x_pareto / PARETO_THRESHOLD)
)

with warnings.catch_warnings():
    warnings.simplefilter("ignore")

    dagum_a, dagum_p, _, dagum_b = burr.fit(
        x_dagum,
        floc=0,
    )


rng = np.random.default_rng(
    BOOTSTRAP_SEED
)

alpha_boot = np.empty(
    BOOTSTRAP_REPLICATIONS
)

a_boot = np.empty(
    BOOTSTRAP_REPLICATIONS
)

b_boot = np.empty(
    BOOTSTRAP_REPLICATIONS
)

p_boot = np.empty(
    BOOTSTRAP_REPLICATIONS
)

with warnings.catch_warnings():
    warnings.simplefilter("ignore")

    for i in range(BOOTSTRAP_REPLICATIONS):

        pareto_sample = rng.choice(
            x_pareto,
            size=n_pareto,
            replace=True,
        )

        dagum_sample = rng.choice(
            x_dagum,
            size=n_dagum,
            replace=True,
        )

        alpha_boot[i] = (
            n_pareto
            / np.sum(
                np.log(
                    pareto_sample / PARETO_THRESHOLD
                )
            )
        )

        fitted_a, fitted_p, _, fitted_b = burr.fit(
            dagum_sample,
            floc=0,
        )

        a_boot[i] = fitted_a
        b_boot[i] = fitted_b
        p_boot[i] = fitted_p


alpha_ci = np.percentile(
    alpha_boot,
    [2.5, 97.5],
)

a_ci = np.percentile(
    a_boot,
    [2.5, 97.5],
)

b_ci = np.percentile(
    b_boot,
    [2.5, 97.5],
)

p_ci = np.percentile(
    p_boot,
    [2.5, 97.5],
)

# The two models use different estimation samples.
# Therefore, AIC and BIC are reported only for transparency.

pareto_loglik = np.sum(
    np.log(pareto_alpha)
    + pareto_alpha * np.log(PARETO_THRESHOLD)
    - (pareto_alpha + 1) * np.log(x_pareto)
)

dagum_loglik = np.sum(
    burr.logpdf(
        x_dagum,
        dagum_a,
        dagum_p,
        loc=0,
        scale=dagum_b,
    )
)

pareto_aic = (
    2 - 2 * pareto_loglik
)

pareto_bic = (
    np.log(n_pareto)
    - 2 * pareto_loglik
)

dagum_aic = (
    6 - 2 * dagum_loglik
)

dagum_bic = (
    3 * np.log(n_dagum)
    - 2 * dagum_loglik
)

dash = "—"

table = pd.DataFrame(
    {
        "Measure": [
            "Estimation sample",
            "Sample size",
            "Arrival rate, λ (events/year)",
            "Threshold, xₘ",
            "Shape parameter, α",
            "Shape parameter, a",
            "Scale parameter, b",
            "Shape parameter, p",
            "Log-likelihood",
            "AIC",
            "BIC",
        ],

        "Pareto": [
            "x ≥ 2×10⁻⁵",
            str(n_pareto),
            f"{n_pareto / 25:.2f}",
            "0.00002",
            (
                f"{pareto_alpha:.6f} "
                f"[{alpha_ci[0]:.6f}, "
                f"{alpha_ci[1]:.6f}]"
            ),
            dash,
            dash,
            dash,
            f"{pareto_loglik:.3f}",
            f"{pareto_aic:.3f}",
            f"{pareto_bic:.3f}",
        ],

        "Dagum": [
            "All positive event losses",
            str(n_dagum),
            f"{n_dagum / 25:.2f}",
            "Not applied",
            dash,
            (
                f"{dagum_a:.6f} "
                f"[{a_ci[0]:.6f}, "
                f"{a_ci[1]:.6f}]"
            ),
            (
                f"{dagum_b:.9f} "
                f"[{b_ci[0]:.9f}, "
                f"{b_ci[1]:.9f}]"
            ),
            (
                f"{dagum_p:.6f} "
                f"[{p_ci[0]:.6f}, "
                f"{p_ci[1]:.6f}]"
            ),
            f"{dagum_loglik:.3f}",
            f"{dagum_aic:.3f}",
            f"{dagum_bic:.3f}",
        ],
    }
)

with pd.ExcelWriter(
    OUTPUT_FILE,
    engine="openpyxl",
) as writer:

    table.to_excel(
        writer,
        sheet_name="Table 1",
        index=False,
    )

    ws = writer.book["Table 1"]

    header_fill = PatternFill(
        "solid",
        fgColor="D9D9D9",
    )

    for cell in ws[1]:
        cell.font = Font(bold=True)
        cell.fill = header_fill
        cell.alignment = Alignment(
            horizontal="center"
        )

    ws.column_dimensions["A"].width = 36
    ws.column_dimensions["B"].width = 38
    ws.column_dimensions["C"].width = 44

    for row in ws.iter_rows(
        min_row=2,
        min_col=2,
        max_col=3,
    ):
        for cell in row:
            cell.alignment = Alignment(
                horizontal="center"
            )

print(table.to_string(index=False))
print()
print("Saved:", OUTPUT_FILE)
